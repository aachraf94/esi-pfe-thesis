% ── Technology logo (graceful) ───────────────────────────────────────────────
% Usage: \techlogo[height]{filename}  — expects a PNG in assets/logos/tech/.
% Renders the logo if the file is present, otherwise a subtle text stub so the
% build never breaks while the logos are still being sourced (see FIGURES_TODO.md).
\newcommand{\techlogo}[2][7mm]{%
    \IfFileExists{assets/logos/tech/#2.png}%
        {\raisebox{-0.35\height}{\includegraphics[height=#1]{#2}}}%
        {\textcolor{black!30}{\scriptsize\ttfamily #2}}%
}
