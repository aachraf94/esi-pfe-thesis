% ── Code listing style ───────────────────────────────────────────────────────
\lstdefinestyle{thesis}{
    backgroundcolor  = \color{backcolour},
    basicstyle       = \ttfamily\small,
    keywordstyle     = \bfseries,
    commentstyle     = \itshape,
    stringstyle      = \normalfont\ttfamily,
    numberstyle      = \tiny\color{codegray},
    breaklines       = true,

    numbers          = left,
    numbersep        = 8pt,
    showstringspaces = false,
    tabsize          = 2,
    captionpos       = b,
}
\lstset{style=thesis}

% ── Caption style ────────────────────────────────────────────────────────────
% Custom label format produces "Table 1.1 ---" so captions need no manual prefix.
\DeclareCaptionLabelFormat{thesisdash}{#1~#2~---}
\captionsetup{
    font       = small,
    labelfont  = bf,
    labelsep   = space,
    justification = centering,
}
\captionsetup[table]{labelformat=thesisdash}
\captionsetup[figure]{labelformat=thesisdash}

% ── Theorem environments ─────────────────────────────────────────────────────
\theoremstyle{definition}
\newtheorem{definition}{Definition}[chapter]
\newtheorem{example}{Example}[chapter]
\theoremstyle{plain}
\newtheorem{theorem}{Theorem}[chapter]

% ── Chapter / section title style ────────────────────────────────────────────
% Non-starred \titlespacing preserves first-paragraph indentation.
\titleformat{\chapter}[display]
    {\normalfont\huge\bfseries}
    {\chaptertitlename~\thechapter}
    {16pt}{\Huge}
\titlespacing{\chapter}{0pt}{-20pt}{40pt}
\assignpagestyle{\chapter}{plain}

\titlespacing{\section}{0pt}{1.5ex plus .5ex minus .2ex}{.5ex plus .1ex}
\titlespacing{\subsection}{0pt}{1.2ex plus .4ex minus .2ex}{.4ex plus .1ex}
\titlespacing{\subsubsection}{0pt}{1ex plus .3ex minus .1ex}{.3ex plus .1ex}
